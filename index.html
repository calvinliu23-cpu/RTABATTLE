function parseHashRoute(){ const hash = location.hash.replace('#/','').replace('#',''); const parts = hash.split('/').filter(Boolean); let serverId = 'default'; let mode = 'public'; if(parts.length === 0){ serverId = 'default'; mode = 'public'; } else if(parts.length === 1){ if(parts[0].toLowerCase() === 'admin'){ // 加了 .toLowerCase() serverId = 'default'; mode = 'admin'; } else { serverId = parts[0]; mode = 'public'; } } else { serverId = parts[0]; mode = parts[1].toLowerCase() === 'admin' ? 'admin' : 'public'; // 加了 .toLowerCase() } return { serverId, mode }; }
